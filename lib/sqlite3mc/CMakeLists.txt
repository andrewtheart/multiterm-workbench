cmake_minimum_required(VERSION 3.24)
project(multiterm_sqlite3mc C)

if(NOT MSVC)
  message(FATAL_ERROR "MultiTerm's packaged SQLite3MC build currently requires MSVC.")
endif()

set(MULTITERM_ARCH "x64" CACHE STRING "Packaged architecture name")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")

add_library(sqlite3mc SHARED upstream/sqlite3mc_amalgamation.c)
set_target_properties(sqlite3mc PROPERTIES
  OUTPUT_NAME "sqlite3mc"
  WINDOWS_EXPORT_ALL_SYMBOLS ON
  RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin/${MULTITERM_ARCH}"
  ARCHIVE_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/lib"
  PDB_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/pdb"
)

target_include_directories(sqlite3mc PRIVATE upstream)
target_compile_options(sqlite3mc PRIVATE /O2 /GS /guard:cf /sdl /W3)
target_link_options(sqlite3mc PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)

target_compile_definitions(sqlite3mc PRIVATE
  _CRT_SECURE_NO_WARNINGS
  CODEC_TYPE=CODEC_TYPE_CHACHA20
  HAVE_CIPHER_AES_128_CBC=0
  HAVE_CIPHER_AES_256_CBC=0
  HAVE_CIPHER_CHACHA20=1
  HAVE_CIPHER_SQLCIPHER=0
  HAVE_CIPHER_RC4=0
  HAVE_CIPHER_ASCON128=0
  HAVE_CIPHER_AEGIS=0
  SQLITE3MC_SECURE_MEMORY=1
  SQLITE_THREADSAFE=1
  SQLITE_DQS=0
  SQLITE_DEFAULT_FOREIGN_KEYS=1
  SQLITE_SECURE_DELETE=1
  SQLITE_OMIT_LOAD_EXTENSION=1
  SQLITE_TEMP_STORE=3
  SQLITE_MAX_ATTACHED=0
  SQLITE_DEFAULT_SYNCHRONOUS=2
  SQLITE_DEFAULT_WAL_SYNCHRONOUS=2
)
